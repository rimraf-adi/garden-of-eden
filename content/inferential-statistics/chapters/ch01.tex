\chapter{Point Estimation of Parameters}

\section{Introduction}

A \emph{population} is described by a probability distribution with one or more unknown constants called \emph{parameters}. Examples: the mean $\mu$ and variance $\sigma^2$ of a normal distribution, the rate $\lambda$ of an exponential distribution, the success probability $p$ of a Bernoulli trial.

We cannot see the parameter. We see a \emph{random sample} $X_1,\dots,X_n$: independent random variables, each with the population distribution. We must use the sample to say something about the parameter.

\begin{definition}[Point estimator and estimate]
\begin{itemize}
  \item A \emph{point estimator} of a parameter $\theta$ is a statistic $\hat\Theta=h(X_1,\dots,X_n)$, a function of the sample meant to be close to $\theta$.
  \item After the data are observed, the number $\hat\theta=h(x_1,\dots,x_n)$ is the \emph{point estimate}.
\end{itemize}
\end{definition}

The difference matters. The estimator is a random variable: it changes from sample to sample. The estimate is one realised number. Statements such as ``it is unbiased'' or ``it has small variance'' are about the distribution of the random variable $\hat\Theta$.

\begin{example}
To estimate a population mean $\mu$ we could use the sample mean $\bar{X}$, the sample median $\tilde{X}$, or the midrange $(X_{\min}+X_{\max})/2$. All three are point estimators of $\mu$. Which is best depends on the population and on what ``best'' means. The next section deals with that.
\end{example}

\section{General Concepts of Point Estimation}

\subsection{Unbiased Estimators}

\begin{definition}[Unbiased estimator]
$\hat\Theta$ is \emph{unbiased} for $\theta$ if $\Exp[\hat\Theta]=\theta$ for every possible value of $\theta$. The \emph{bias} is $\bias(\hat\Theta)=\Exp[\hat\Theta]-\theta$.
\end{definition}

Unbiased means correct on average over repeated samples. The estimator does not systematically overshoot or undershoot.

\begin{theorem}[Standard unbiased estimators]\label{thm:unbiased}
Let $X_1,\dots,X_n$ be a random sample from a distribution with mean $\mu$ and variance $\sigma^2$. Then
\begin{enumerate}[label=(\roman*)]
  \item $\bar{X}$ is unbiased for $\mu$;
  \item $S^2=\dfrac{1}{n-1}\sum_{i=1}^n(X_i-\bar{X})^2$ is unbiased for $\sigma^2$;
  \item for Bernoulli trials with success probability $p$, and $X$ the number of successes, $\hat{P}=X/n$ is unbiased for $p$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) $\Exp[\bar{X}]=\frac1n\sum\Exp[X_i]=\mu$.

(iii) $\Exp[X]=np$, so $\Exp[\hat{P}]=p$.

(ii) Expand the sum of squares about the true mean:
\[ \sum(X_i-\bar{X})^2=\sum(X_i-\mu)^2-n(\bar{X}-\mu)^2. \]
Take expectations, using $\Var[\bar{X}]=\sigma^2/n$:
\[ \Exp\Big[\sum(X_i-\bar{X})^2\Big]=n\sigma^2-n\cdot\frac{\sigma^2}n=(n-1)\sigma^2. \]
Divide by $n-1$ to get $\Exp[S^2]=\sigma^2$.
\end{proof}

\begin{intuition}
The sum of squares around $\bar{X}$ is smaller than the sum around the true $\mu$, because $\bar{X}$ is the point that minimises the sum of squares for this very sample.
\begin{itemize}
  \item Dividing by $n$ would underestimate $\sigma^2$ on average, by the factor $(n-1)/n$.
  \item Dividing by $n-1$ corrects this exactly.
  \item The estimator $\frac1n\sum(X_i-\bar{X})^2$ is biased, with $\bias=-\sigma^2/n$.
\end{itemize}
\end{intuition}

\begin{example}
Four measurements: $12.1,\ 11.8,\ 12.4,\ 12.3$.
\begin{itemize}
  \item $\bar{x}=12.15$.
  \item Deviations: $-0.05,\ -0.35,\ 0.25,\ 0.15$. Squares: $0.0025,\ 0.1225,\ 0.0625,\ 0.0225$. Sum: $0.21$.
  \item $s^2=0.21/3=0.07$ and $s=0.265$.
\end{itemize}
These are point estimates of $\mu$ and $\sigma$.
\end{example}

\begin{nuance}{unbiased is not always sensible}
An unbiased estimator may not exist. When it exists, it may be poor.

If $X$ is one observation from a Poisson distribution, the only unbiased estimator of $e^{-2\lambda}$ is $(-1)^X$. It takes the values $\pm1$ for a quantity that lies in $(0,1)$.

Unbiasedness is desirable, not required. Section~\ref{sec:mse} balances bias against variance.
\end{nuance}

\subsection{Proof that $S$ is a Biased Estimator of $\sigma$ (CD Only)}

$S^2$ is unbiased for $\sigma^2$, but $S=\sqrt{S^2}$ is \emph{not} unbiased for $\sigma$.

\begin{theorem}\label{thm:sbiased}
If $\sigma^2>0$ and the sample is not degenerate, $\Exp[S]<\sigma$.
\end{theorem}

\begin{proof}
The square root is strictly concave. By Jensen's inequality, $\Exp[\sqrt{S^2}]<\sqrt{\Exp[S^2]}=\sigma$. The inequality is strict because $S^2$ is not constant.
\end{proof}

For a normal sample the bias is explicit: $\Exp[S]=c_4(n)\,\sigma$, where
\[ c_4(n)=\sqrt{\dfrac2{n-1}}\,\dfrac{\Gamma(n/2)}{\Gamma((n-1)/2)}. \]
For example $c_4(5)\approx0.940$, $c_4(10)\approx0.973$, $c_4(25)\approx0.990$. The bias is small unless $n$ is small, and is usually ignored. Control charts (Chapter~10) do use $c_4$ to correct it.

\subsection{Variance of a Point Estimator}

Among several unbiased estimators of the same parameter, prefer the one with the smallest variance. It varies least from sample to sample.

\begin{definition}[MVUE]
Among all unbiased estimators of $\theta$, the one with the smallest variance is the \emph{minimum variance unbiased estimator} (MVUE).
\end{definition}

\begin{example}[Mean versus median]
For a sample from $\Normal(\mu,\sigma^2)$, both $\bar{X}$ and the sample median $\tilde{X}$ are unbiased for $\mu$ (the normal is symmetric).
\begin{itemize}
  \item $\Var[\bar{X}]=\sigma^2/n$.
  \item For large $n$, $\Var[\tilde{X}]\approx\frac\pi2\cdot\frac{\sigma^2}n\approx1.57\,\sigma^2/n$.
\end{itemize}
The median needs about $57\%$ more data for the same accuracy.
\end{example}

\begin{theorem}[$\bar{X}$ is the MVUE for the normal mean]\label{thm:mvue}
If $X_1,\dots,X_n$ is a random sample from $\Normal(\mu,\sigma^2)$, then $\bar{X}$ is the MVUE of $\mu$.
\end{theorem}

This follows from the Cram\'er--Rao inequality, proved in Chapter~12. For non-normal populations the answer changes. For a uniform distribution on $(\mu-a,\mu+a)$, the midrange has variance of order $1/n^2$. That is far smaller than $\Var[\bar{X}]=a^2/(3n)$.

\subsection{Standard Error: Reporting a Point Estimator}

A point estimate alone is incomplete. We also need a measure of its precision.

\begin{definition}[Standard error]
The \emph{standard error} of an estimator $\hat\Theta$ is its standard deviation, $\sigma_{\hat\Theta}=\sqrt{\Var[\hat\Theta]}$. If it depends on unknown parameters, replace them by estimates. The result is the \emph{estimated standard error}, written $\hat\sigma_{\hat\Theta}$ or $\se(\hat\Theta)$.
\end{definition}

For the sample mean, $\sigma_{\bar{X}}=\sigma/\sqrt n$. Since $\sigma$ is usually unknown, the estimated standard error is $\se(\bar{X})=s/\sqrt n$. Standard practice is to report $\hat\theta\ \pm\ \se(\hat\theta)$.

\begin{example}
Thermal conductivity of $n=10$ specimens: $\bar{x}=41.924$, $s=0.284$.
\begin{itemize}
  \item $\se(\bar{x})=s/\sqrt{10}=0.0898$. Report $41.924\pm0.0898$.
  \item For a normal population, about $95\%$ of the time $\bar{X}$ is within $2\sigma_{\bar{X}}$ of $\mu$ (Section~\ref{sec:sampmeans}).
\end{itemize}
\end{example}

\begin{pitfall}{standard error is not standard deviation}
\begin{itemize}
  \item The standard deviation $s$ measures the spread of individual observations.
  \item The standard error $s/\sqrt n$ measures the spread of the \emph{mean} over repeated samples. It shrinks as $n$ grows.
\end{itemize}
Reporting $\pm s$ when you mean $\pm\se$ overstates the uncertainty of the mean by a factor $\sqrt n$.
\end{pitfall}

\begin{example}[Thermal conductivity of Armco iron]
Ten measurements at $100^\circ$F and $550$ W give $\bar{x}=41.924$, $s=0.284$.
\begin{itemize}
  \item $\se=s/\sqrt{10}=0.0898$, about $0.2\%$ of the estimate. The estimate is precise.
  \item If conductivity is normal, $2\,\widehat{\se}=0.1796$. We are highly confident the true mean lies in $41.924\pm0.1796$, that is $(41.744,\,42.104)$.
  \item Even if the estimator is not normal, as long as it is unbiased, the estimate is off by four standard errors or more at most about $6\%$ of the time (Chebyshev). So a very cautious statement: the parameter is within four standard errors of the estimate.
\end{itemize}
\end{example}

\subsection{Bootstrap Estimate of the Standard Error (CD Only)}

For many estimators there is no formula for the standard error. The \emph{bootstrap} estimates it by resampling the data.

\begin{definition}[Bootstrap standard error]
Given data $x_1,\dots,x_n$ and an estimator $\hat\theta=h(x_1,\dots,x_n)$:
\begin{enumerate}[label=(\arabic*)]
  \item Draw a \emph{bootstrap sample} $x_1^*,\dots,x_n^*$: $n$ values sampled \emph{with replacement} from the data.
  \item Compute $\hat\theta^*=h(x_1^*,\dots,x_n^*)$.
  \item Repeat steps 1--2 $B$ times to get $\hat\theta^*_1,\dots,\hat\theta^*_B$.
  \item The bootstrap standard error is their sample standard deviation:
  \[ \se_{\mathrm{boot}}(\hat\theta)=\sqrt{\frac1{B-1}\sum_{b=1}^B\big(\hat\theta^*_b-\bar{\theta}^*\big)^2},\qquad\bar{\theta}^*=\frac1B\sum_b\hat\theta^*_b. \]
\end{enumerate}
Typically $B$ between $200$ and $1000$ is enough for a standard error.
\end{definition}

The idea: the observed data are our best picture of the population, so sampling from the data mimics sampling from the population. It works for the median, a trimmed mean, a correlation, a ratio, or any statistic. Chapter~13 gives more detail.

\begin{example}
Data: $2,\ 5,\ 9$. The bootstrap sample $(5,5,9)$ has mean $6.33$. The sample $(2,2,2)$ has mean $2$. After many repeats, the standard deviation of the bootstrap means approaches $\sqrt{\sum(x_i-\bar{x})^2/n^2}$, which is about $1.7$ for $n=3$.
\end{example}

\subsection{Mean Square Error of an Estimator}\label{sec:mse}

Unbiased estimators are not always best. A small bias can buy a large cut in variance. We need one criterion that counts both.

\begin{definition}[Mean square error]
$\MSE(\hat\Theta)=\Exp\big[(\hat\Theta-\theta)^2\big]$: the expected squared distance from the target.
\end{definition}

\begin{theorem}[Bias--variance decomposition]\label{thm:biasvar}
\[ \MSE(\hat\Theta)=\Var(\hat\Theta)+\big[\bias(\hat\Theta)\big]^2. \]
\end{theorem}

\begin{proof}
Write $\hat\Theta-\theta=(\hat\Theta-\Exp[\hat\Theta])+(\Exp[\hat\Theta]-\theta)$ and square. The cross term has expectation $2(\Exp[\hat\Theta]-\theta)\Exp(\hat\Theta-\Exp[\hat\Theta])=0$. What remains is the variance plus the squared bias.
\end{proof}

For an unbiased estimator, the MSE equals the variance. For biased ones, the decomposition shows the trade-off.

\begin{example}[A biased estimator can beat an unbiased one]\label{ex:mseexample}
Let $X_1,\dots,X_n$ be normal with variance $\sigma^2$. Consider $\hat\sigma^2_c=c\sum(X_i-\bar{X})^2$.
\begin{itemize}
  \item $\sum(X_i-\bar{X})^2/\sigma^2\sim\chi^2_{n-1}$, which has mean $n-1$ and variance $2(n-1)$.
  \item So $\MSE=\sigma^4\big[2c^2(n-1)+((n-1)c-1)^2\big]$. It is minimised at $c=1/(n+1)$.
  \item The unbiased choice $c=1/(n-1)$ has $\MSE=2\sigma^4/(n-1)$.
  \item $c=1/(n+1)$ gives the smaller $\MSE=2\sigma^4/(n+1)$.
\end{itemize}
\end{example}

\begin{definition}[Relative efficiency, consistency]
\begin{itemize}
  \item The \emph{relative efficiency} of $\hat\Theta_2$ to $\hat\Theta_1$ is $\MSE(\hat\Theta_1)/\MSE(\hat\Theta_2)$. If it is less than $1$, $\hat\Theta_1$ is more efficient: it has the smaller mean square error.
  \item An estimator is \emph{consistent} if $\hat\Theta_n\to\theta$ in probability as $n\to\infty$. Sufficient condition: $\bias\to0$ and $\Var\to0$, so that $\MSE\to0$.
\end{itemize}
\end{definition}

\section{Methods of Point Estimation}

So far we judged estimators proposed ad hoc. Two systematic recipes produce estimators from the distribution.

\subsection{Method of Moments}

\begin{definition}[Population and sample moments]
The $k$-th \emph{population moment} is $\Exp(X^k)$. The $k$-th \emph{sample moment} is $\frac1n\sum_{i=1}^nX_i^k$.
\end{definition}

\begin{definition}[Method of moments]
Suppose the distribution has $m$ unknown parameters $\theta_1,\dots,\theta_m$.
\begin{enumerate}
  \item Equate the first $m$ population moments (functions of the parameters) to the first $m$ sample moments.
  \item Solve the $m$ equations for the parameters.
\end{enumerate}
The solutions are the \emph{moment estimators}.
\end{definition}

\begin{example}[Exponential]\label{ex:momexp}
For the exponential with rate $\lambda$, $\Exp[X]=1/\lambda$. Setting this equal to $\bar{X}$ gives $\hat\lambda=1/\bar{X}$.
\end{example}

\begin{example}[Normal]
$\Exp[X]=\mu$ and $\Exp[X^2]=\sigma^2+\mu^2$. Equating to $\bar{X}$ and $\frac1n\sum X_i^2$:
\[ \hat\mu=\bar{X},\qquad\hat\sigma^2=\frac1n\sum X_i^2-\bar{X}^2=\frac1n\sum(X_i-\bar{X})^2. \]
This is the \emph{biased} version of the sample variance.
\end{example}

\begin{example}[Gamma]
For a gamma distribution with shape $r$ and rate $\lambda$: $\Exp[X]=r/\lambda$ and $\Var[X]=r/\lambda^2$. Use the mean and the variance $\hat v=\frac1n\sum(X_i-\bar{X})^2$:
\[ \hat r=\bar{X}^2/\hat v,\qquad\hat\lambda=\bar{X}/\hat v. \]
\end{example}

\begin{example}[Uniform]
For a uniform distribution on $(0,\theta)$, $\Exp[X]=\theta/2$, so $\hat\theta=2\bar{X}$. This is unbiased but can be silly. For the sample $(1,2,9)$, $\hat\theta=8$, smaller than the observed $9$, which is impossible. The maximum likelihood estimator below avoids this.
\end{example}

Moment estimators are simple and usually consistent. But they need not be efficient, and they can fall outside the parameter space.

\begin{example}[Failure-time data]
Eight electronic modules, tested at elevated temperature, have exponential failure times (hours) with $\bar{x}=21.65$ and $\sum x_i^2=6639.40$.
\begin{itemize}
  \item Exponential rate by moments: $\hat\lambda=1/\bar{x}=0.0462$.
  \item Gamma fit by moments: $\hat{r}=1.30$ and $\hat\lambda=0.0599$.
  \item $\hat{r}$ is slightly above $1$ (where the gamma reduces to the exponential), so either distribution is plausible.
\end{itemize}
\end{example}

\subsection{Method of Maximum Likelihood}

\begin{definition}[Likelihood function]
Let $X_1,\dots,X_n$ be a random sample from a distribution with mass or density function $f(x;\theta)$. The \emph{likelihood function} of the observed data is
\[ L(\theta)=f(x_1;\theta)\,f(x_2;\theta)\cdots f(x_n;\theta)=\prod_{i=1}^nf(x_i;\theta), \]
seen as a function of $\theta$ with the data fixed. The \emph{maximum likelihood estimator} (MLE) is the value $\hat\theta$ that maximises $L(\theta)$.
\end{definition}

\begin{intuition}
For discrete data, $L(\theta)$ is the probability, under parameter $\theta$, of getting exactly the data we saw. The MLE picks the parameter value under which the observed data were most probable. It makes the data least surprising.
\end{intuition}

The logarithm is increasing, so maximising $L$ is the same as maximising the \emph{log-likelihood} $\ell(\theta)=\ln L(\theta)=\sum\ln f(x_i;\theta)$. It turns products into sums, which are easier to differentiate.

\textbf{Recipe.}
\begin{enumerate}
  \item Write $\ell(\theta)$.
  \item Solve $\ell'(\theta)=0$ (the \emph{likelihood equation}).
  \item Check that the solution is a maximum.
\end{enumerate}

\begin{example}[Bernoulli]\label{ex:mlebern}
For $n$ trials with $x$ successes: $L(p)=p^x(1-p)^{n-x}$ and $\ell(p)=x\ln p+(n-x)\ln(1-p)$. Set $\ell'(p)=x/p-(n-x)/(1-p)=0$. This gives $\hat{p}=x/n$, the sample proportion.
\end{example}

\begin{example}[Exponential]\label{ex:mleexp}
For $f(x;\lambda)=\lambda e^{-\lambda x}$: $\ell(\lambda)=n\ln\lambda-\lambda\sum x_i$. Set $\ell'(\lambda)=n/\lambda-\sum x_i=0$. This gives $\hat\lambda=n/\sum x_i=1/\bar{x}$, the same as the moment estimator.
\end{example}

\begin{example}[Normal]\label{ex:mlenormal}
For $\Normal(\mu,\sigma^2)$:
\[ \ell(\mu,\sigma^2)=-\frac n2\ln(2\pi\sigma^2)-\frac1{2\sigma^2}\sum(x_i-\mu)^2. \]
\begin{itemize}
  \item $\partial\ell/\partial\mu=\frac1{\sigma^2}\sum(x_i-\mu)=0$ gives $\hat\mu=\bar{x}$.
  \item $\partial\ell/\partial\sigma^2=-\frac n{2\sigma^2}+\frac1{2\sigma^4}\sum(x_i-\mu)^2=0$ gives $\hat\sigma^2=\frac1n\sum(x_i-\bar{x})^2$.
\end{itemize}
So the MLE of the variance is the \emph{biased} estimator with divisor $n$.
\end{example}

\begin{example}[Poisson]
For $f(x;\lambda)=e^{-\lambda}\lambda^x/x!$: $\ell(\lambda)=-n\lambda+(\sum x_i)\ln\lambda-\sum\ln x_i!$. So $\hat\lambda=\bar{x}$.
\end{example}

\begin{example}[Uniform: a likelihood without a stationary point]\label{ex:mleunif}
For $f(x;\theta)=1/\theta$ on $0\le x\le\theta$: $L(\theta)=\theta^{-n}$ if $\theta\ge\max x_i$, and $0$ otherwise. $\theta^{-n}$ is decreasing, so the maximum is at the smallest allowed value: $\hat\theta=\max(x_1,\dots,x_n)$. Calculus does not apply here. Look at the shape of $L$ directly.
\end{example}

\begin{theorem}[Invariance property]\label{thm:invariance}
If $\hat\theta$ is the MLE of $\theta$, then for any function $g$, the MLE of $g(\theta)$ is $g(\hat\theta)$.
\end{theorem}

\begin{example}
$\hat\sigma^2=\frac1n\sum(x_i-\bar{x})^2$ is the MLE of $\sigma^2$ for a normal sample. So the MLE of $\sigma$ is $\hat\sigma=\sqrt{\hat\sigma^2}$. For the exponential, the MLE of the mean $1/\lambda$ is $\bar{x}$, and the MLE of $\Prob(X>t)=e^{-\lambda t}$ is $e^{-t/\bar{x}}$.
\end{example}

\begin{theorem}[Large-sample properties of the MLE]\label{thm:mleasymp}
Under mild regularity conditions, as $n\to\infty$ the MLE $\hat\Theta$ is
\begin{enumerate}[label=(\roman*)]
  \item asymptotically unbiased: $\Exp[\hat\Theta]\to\theta$;
  \item consistent;
  \item asymptotically normal: $\hat\Theta\approx\Normal\big(\theta,\ 1/(n\Fish(\theta))\big)$. The variance equals the Cram\'er--Rao lower bound. Here $\Fish(\theta)=\Exp\big[-\partial^2\ln f(X;\theta)/\partial\theta^2\big]$ is the Fisher information of one observation;
  \item therefore asymptotically the MVUE: no consistent estimator has smaller large-sample variance.
\end{enumerate}
\end{theorem}

Fisher information and the Cram\'er--Rao bound are developed in Chapter~12. Practical use: the standard error of an MLE can be estimated by $1/\sqrt{n\Fish(\hat\theta)}$. This is the reciprocal square root of the curvature of the log-likelihood at its peak.

\begin{example}
\begin{itemize}
  \item Bernoulli: $\Fish(p)=1/[p(1-p)]$, so $\se(\hat{p})\approx\sqrt{\hat{p}(1-\hat{p})/n}$. This is the familiar standard error of a proportion.
  \item Exponential: $\Fish(\lambda)=1/\lambda^2$, so $\se(\hat\lambda)\approx\hat\lambda/\sqrt n$.
\end{itemize}
\end{example}

\begin{interview}{when the MLE is awkward}
\begin{itemize}
  \item Likelihood equations often have no closed form (gamma shape, logistic regression). They are solved numerically, by Newton's method or EM.
  \item The MLE can be biased in small samples (normal variance).
  \item It can fail to be a stationary point (uniform).
  \item It can be non-unique or even infinite (logistic regression with perfectly separated classes).
\end{itemize}
Know the Bernoulli, exponential, Poisson and normal cases by heart.
\end{interview}

\subsection{Bayesian Estimation of Parameters (CD Only)}

The methods so far treat $\theta$ as an unknown constant. The Bayesian approach treats $\theta$ as a random variable with a \emph{prior} distribution $f(\theta)$, which says what is known before the data. Bayes' theorem updates it to the \emph{posterior} after seeing $x_1,\dots,x_n$:
\[ f(\theta\mid x_1,\dots,x_n)=\frac{f(x_1,\dots,x_n\mid\theta)\,f(\theta)}{\int f(x_1,\dots,x_n\mid\theta')f(\theta')\,d\theta'}\ \propto\ L(\theta)\,f(\theta). \]
The \emph{Bayes estimator} of $\theta$ is the mean of the posterior. It minimises the expected squared error.

\begin{example}[Bernoulli with a beta prior]\label{ex:betabin}
Let $X\sim\mathrm{Binomial}(n,p)$ with prior $p\sim\mathrm{Beta}(a,b)$, density $\propto p^{a-1}(1-p)^{b-1}$. Multiply by the likelihood $p^x(1-p)^{n-x}$. The posterior is $\mathrm{Beta}(a+x,\ b+n-x)$. The Bayes estimator is the posterior mean
\[ \hat{p}_B=\frac{a+x}{a+b+n}=\frac{n}{a+b+n}\cdot\frac xn+\frac{a+b}{a+b+n}\cdot\frac a{a+b}. \]
This is a weighted average of the sample proportion and the prior mean. As $n$ grows, the data dominate and $\hat{p}_B\to x/n$.
\end{example}

\begin{example}[Normal mean with a normal prior]
Let $X_1,\dots,X_n$ be $\Normal(\mu,\sigma^2)$ with $\sigma^2$ known, and prior $\mu\sim\Normal(\mu_0,\sigma_0^2)$. The posterior is normal with
\[ \text{mean }\ \frac{(\sigma^2/n)\mu_0+\sigma_0^2\bar{x}}{\sigma^2/n+\sigma_0^2},\qquad\text{variance }\ \frac{(\sigma^2/n)\sigma_0^2}{\sigma^2/n+\sigma_0^2}. \]
The estimate pulls $\bar{x}$ toward the prior mean. The pull is stronger when the prior is precise (small $\sigma_0^2$) or the data are few.
\end{example}

\section{Sampling Distributions}

\begin{definition}[Sampling distribution]
A statistic is a function of the sample, so it is a random variable. Its probability distribution is its \emph{sampling distribution}.
\end{definition}

Every estimator and test statistic in this book has a sampling distribution. All of inference rests on knowing it.

\begin{example}
A population has the equally likely values $1,3,5$. Take samples of size $2$ with replacement: nine equally likely pairs.
\begin{itemize}
  \item $(1,1)\to1$; $(1,3),(3,1)\to2$; $(1,5),(5,1),(3,3)\to3$; $(3,5),(5,3)\to4$; $(5,5)\to5$.
  \item So $\bar{X}$ has probabilities $1/9,\ 2/9,\ 3/9,\ 2/9,\ 1/9$ on $1,2,3,4,5$.
  \item Its mean is $3$ (the population mean). Its variance is $4/3=\sigma^2/2$, where $\sigma^2=8/3$.
\end{itemize}
\end{example}

\section{Sampling Distribution of Means}\label{sec:sampmeans}

\begin{theorem}[Distribution of $\bar{X}$]\label{thm:dist-xbar}
Let $X_1,\dots,X_n$ be a random sample with mean $\mu$ and variance $\sigma^2$. Then $\Exp[\bar{X}]=\mu$ and $\Var[\bar{X}]=\sigma^2/n$. If the population is normal, $\bar{X}$ is exactly $\Normal(\mu,\sigma^2/n)$.
\end{theorem}

\begin{theorem}[Central limit theorem]\label{thm:clt-text}
If $X_1,\dots,X_n$ is a random sample from a population with mean $\mu$ and finite variance $\sigma^2$, then for large $n$
\[ Z=\frac{\bar{X}-\mu}{\sigma/\sqrt n}\ \text{ is approximately }\Normal(0,1),\qquad\text{that is, }\bar{X}\approx\Normal\Big(\mu,\frac{\sigma^2}n\Big). \]
The approximation improves as $n$ grows.
\begin{itemize}
  \item For most populations it is good when $n\ge30$.
  \item For a nearly symmetric population, smaller $n$ is enough.
  \item For a strongly skewed population, larger $n$ may be needed.
\end{itemize}
\end{theorem}

\begin{example}
Resistors have mean $\mu=100\ \Omega$ and standard deviation $\sigma=10\ \Omega$. For a sample of $n=25$, what is the probability that the average resistance is below $95\ \Omega$?

$\sigma_{\bar{X}}=10/\sqrt{25}=2$, so $\Prob(\bar{X}<95)\approx\Prob\big(Z<(95-100)/2\big)=\Prob(Z<-2.5)=0.0062$.
\end{example}

\begin{theorem}[Difference of two sample means]\label{thm:diffmeans}
Take independent random samples of sizes $n_1,n_2$ from populations with means $\mu_1,\mu_2$ and variances $\sigma_1^2,\sigma_2^2$. Then $\bar{X}_1-\bar{X}_2$ has mean $\mu_1-\mu_2$ and variance $\sigma_1^2/n_1+\sigma_2^2/n_2$. Also
\[ Z=\frac{(\bar{X}_1-\bar{X}_2)-(\mu_1-\mu_2)}{\sqrt{\sigma_1^2/n_1+\sigma_2^2/n_2}} \]
is exactly $\Normal(0,1)$ if both populations are normal, and approximately so for large $n_1,n_2$.
\end{theorem}

\begin{example}
Two machines fill cartons. Machine 1: $\mu_1=500$, $\sigma_1=6$. Machine 2: $\mu_2=497$, $\sigma_2=5$. With $n_1=n_2=36$:
\begin{itemize}
  \item $\bar{X}_1-\bar{X}_2$ has mean $3$ and standard deviation $\sqrt{36/36+25/36}=1.30$.
  \item $\Prob(\bar{X}_1>\bar{X}_2)=\Prob(Z>(0-3)/1.30)=\Prob(Z>-2.30)=0.989$.
\end{itemize}
\end{example}

\begin{example}[Further textbook examples]
\begin{itemize}
  \item \emph{Uniform.} If $X\sim U(4,6)$, then $\mu=5$ and $\sigma^2=(6-4)^2/12=1/3$. For $n=40$, the CLT gives $\bar{X}\approx\Normal(5,\ \sigma^2/n=1/120)$.
  \item \emph{Two samples.} A jet-engine component has life with mean $5000$ h and $\sigma=40$ (old process). After improvement: mean $5050$ h and $\sigma=30$ (new process). Samples: $n_1=16$ old, $n_2=25$ new. Then $\bar{X}_2-\bar{X}_1\sim\Normal\big(50,\ 30^2/25+40^2/16=36+100=136\big)$. So $\Prob(\bar{X}_2-\bar{X}_1\ge25)=\Prob\big(Z\ge(25-50)/\sqrt{136}=-2.14\big)=0.9838$.
  \item \emph{Likelihood and sample size.} For the exponential failure data, the log-likelihood peaks at $\hat\lambda=0.0462$ but is quite flat for $n=8$, so $\lambda$ is not estimated precisely. With $\bar{x}=21.65$ held fixed, it becomes steeper (more sharply peaked) for $n=20$ and $n=40$. For the gamma model, numerical maximum likelihood gives about $\hat{r}=1.75$ and $\hat\lambda=0.08$.
\end{itemize}
\end{example}

\begin{summarybox}
\begin{itemize}
  \item An estimator is a statistic. Its value is an estimate. Judge estimators by bias, variance and $\MSE=\Var+\bias^2$.
  \item $\bar{X}$, $S^2$ and $\hat{P}$ are unbiased. $S$ is slightly biased low. The mean beats the median for normal data (the median has about $57\%$ larger variance).
  \item Report estimates with their standard errors. The bootstrap gives standard errors without a formula.
  \item Method of moments equates sample and population moments. Maximum likelihood maximises $L(\theta)=\prod f(x_i;\theta)$. It is invariant under transformations, and asymptotically unbiased, normal and efficient.
  \item Bayesian estimation combines a prior with the likelihood. The posterior mean is a weighted average of the data estimate and the prior mean.
  \item $\bar{X}\approx\Normal(\mu,\sigma^2/n)$ (exactly for normal populations, by the CLT otherwise). $\bar{X}_1-\bar{X}_2$ has variance $\sigma_1^2/n_1+\sigma_2^2/n_2$.
\end{itemize}
\end{summarybox}

\section{Exercises}

\begin{exercise}
Let $X_1,X_2,\dots,X_7$ be a random sample from a population with mean $\mu$ and variance $\sigma^2$. Consider $\hat\Theta_1=(X_1+X_2+\dots+X_7)/7$ and $\hat\Theta_2=(2X_1-X_6+X_4)/2$. Is either biased? Find the variance of each and say which is preferable.
\end{exercise}

\begin{exercise}
Show that $\sum_{i}(X_i-\bar{X})^2=\sum_i X_i^2-n\bar{X}^2$ and that $\frac1n\sum(X_i-\bar{X})^2$ has bias $-\sigma^2/n$. Find the MSE of this biased estimator for a normal sample and compare it with the MSE of $S^2$.
\end{exercise}

\begin{exercise}
For a random sample of size $n$ from a Poisson distribution with parameter $\lambda$, find the method of moments estimator and the maximum likelihood estimator of $\lambda$, their bias and variance, and the estimated standard error.
\end{exercise}

\begin{exercise}
Find the maximum likelihood estimator of $\theta$ for the density $f(x;\theta)=\theta x^{\theta-1}$, $0<x<1$, $\theta>0$, and the moment estimator. Which would you prefer when $n=3$ and the data are $0.2,0.5,0.9$?
\end{exercise}

\begin{exercise}
A sample of size $n=20$ from an exponential distribution has $\sum x_i=64$. Find the MLE of $\lambda$, of the mean $1/\lambda$ and of $\Prob(X>5)$ (use invariance), and the estimated standard error of $\hat\lambda$.
\end{exercise}

\begin{exercise}
Show that for $\hat\sigma^2_c=c\sum(X_i-\bar{X})^2$ the MSE is $\sigma^4[2c^2(n-1)+((n-1)c-1)^2]$ and verify the minimiser $c=1/(n+1)$.
\end{exercise}

\begin{exercise}
In a quality study $x=12$ defectives are found in $n=100$ items. Using a $\mathrm{Beta}(2,18)$ prior, compute the posterior distribution of $p$ and the Bayes estimate. Compare with the MLE.
\end{exercise}

\begin{exercise}
The scores of a test have $\mu=75$, $\sigma=12$. For a random sample of $36$ students find $\Prob(\bar{X}>78)$, $\Prob(72<\bar{X}<77)$, and the value $c$ such that $\Prob(\bar{X}>c)=0.05$.
\end{exercise}

\begin{exercise}
Write a short program that draws $B=1000$ bootstrap samples from the data $14,\ 18,\ 11,\ 16,\ 25,\ 9,\ 13$ and computes the bootstrap standard error of (a) the mean and (b) the median. Compare (a) with $s/\sqrt n$.
\end{exercise}
