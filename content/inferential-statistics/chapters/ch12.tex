\chapter{Supplement: Estimation Theory}

\begin{nuance}{place in the course}
This supplementary chapter proves the results that Chapter~1 states without proof: why the sample mean is the MVUE for a normal mean, how much precision is possible at all (the Cram\'er--Rao bound), why the maximum likelihood estimator is asymptotically optimal, and what it means for a statistic to carry all the information about a parameter (sufficiency). It is written for readers who want the mathematics behind Chapter~1; it is not required for the textbook's problems.
\end{nuance}

\section{Score function and Fisher information}

Let $X$ have density (or mass function) $f(x;\theta)$ with a scalar parameter $\theta$. Assume the usual \emph{regularity conditions}: the support of $f$ does not depend on $\theta$, $f$ is twice differentiable in $\theta$, and differentiation and integration can be interchanged.

\begin{definition}[Score and Fisher information]
The \emph{score} is $U(\theta)=\dfrac{\partial}{\partial\theta}\ln f(X;\theta)$. The \emph{Fisher information} of one observation is $\Fish(\theta)=\Var\big(U(\theta)\big)$. For a random sample of size $n$ the information is $n\Fish(\theta)$.
\end{definition}

\begin{theorem}[Properties of the score]\label{thm:score}
Under regularity, $\Exp[U(\theta)]=0$ and
\[ \Fish(\theta)=\Exp\Big[\Big(\frac{\partial\ln f}{\partial\theta}\Big)^2\Big]=-\Exp\Big[\frac{\partial^2\ln f}{\partial\theta^2}\Big]. \]
\end{theorem}

\begin{proof}
Since $\int f(x;\theta)\,dx=1$, differentiating gives $0=\int\partial_\theta f\,dx=\int(\partial_\theta\ln f)f\,dx=\Exp[U]$. Differentiating $\int(\partial_\theta\ln f)f\,dx=0$ once more: $0=\int(\partial_\theta^2\ln f)f\,dx+\int(\partial_\theta\ln f)\partial_\theta f\,dx=\Exp[\partial_\theta^2\ln f]+\Exp[U^2]$. Since $\Exp U=0$, $\Var U=\Exp U^2=-\Exp[\partial_\theta^2\ln f]$.
\end{proof}

The second form shows that information is the expected \emph{curvature} of the log-likelihood: a sharply peaked likelihood (large curvature) determines $\theta$ precisely.

\begin{example}[Standard cases]\label{ex:fisher}
\begin{itemize}
  \item Bernoulli$(p)$: $\ln f=x\ln p+(1-x)\ln(1-p)$, $\partial^2\ln f/\partial p^2=-x/p^2-(1-x)/(1-p)^2$, so $\Fish(p)=1/p+1/(1-p)=\dfrac1{p(1-p)}$.
  \item Poisson$(\lambda)$: $\ln f=x\ln\lambda-\lambda-\ln x!$; $\Fish(\lambda)=\Exp[X]/\lambda^2=1/\lambda$.
  \item Exponential with rate $\lambda$: $\ln f=\ln\lambda-\lambda x$; $\Fish(\lambda)=1/\lambda^2$.
  \item $\Normal(\mu,\sigma^2)$ with $\sigma^2$ known: $\Fish(\mu)=1/\sigma^2$; with $\mu$ known, $\Fish(\sigma^2)=1/(2\sigma^4)$.
\end{itemize}
\end{example}

\section{The Cram\'er--Rao lower bound}

\begin{theorem}[Cram\'er--Rao]\label{thm:cramerrao}
Let $X_1,\dots,X_n$ be a random sample and let $\hat\Theta$ be an unbiased estimator of $\theta$. Under regularity,
\[ \Var(\hat\Theta)\ \ge\ \frac1{n\Fish(\theta)}. \]
More generally, for an estimator with $\Exp\hat\Theta=\theta+b(\theta)$, $\Var(\hat\Theta)\ge[1+b'(\theta)]^2/(n\Fish(\theta))$.
\end{theorem}

\begin{proof}
Let $U_n=\sum_i\partial_\theta\ln f(X_i;\theta)$ be the score of the whole sample, with $\Exp U_n=0$ and $\Var U_n=n\Fish(\theta)$. For unbiased $\hat\Theta$, $\theta=\Exp\hat\Theta=\int\hat\theta(\mathbf{x})f(\mathbf{x};\theta)\,d\mathbf{x}$; differentiating with respect to $\theta$ gives $1=\int\hat\theta\,\partial_\theta f\,d\mathbf{x}=\Exp[\hat\Theta U_n]=\Cov(\hat\Theta,U_n)$ (since $\Exp U_n=0$). By the Cauchy--Schwarz inequality for covariances, $1=\Cov(\hat\Theta,U_n)^2\le\Var(\hat\Theta)\Var(U_n)=\Var(\hat\Theta)\,n\Fish(\theta)$.
\end{proof}

\begin{definition}[Efficient estimator]
An unbiased estimator whose variance equals the Cram\'er--Rao bound is \emph{efficient}. The \emph{efficiency} of an unbiased estimator is the ratio of the bound to its variance (at most $1$).
\end{definition}

\begin{theorem}[$\bar{X}$ is the MVUE for a normal mean]\label{thm:mvue-proof}
For $X_i\sim\Normal(\mu,\sigma^2)$ with $\sigma^2$ known, $\bar{X}$ is efficient: $\Var\bar{X}=\sigma^2/n=1/(n\Fish(\mu))$. Hence $\bar{X}$ is the MVUE of $\mu$ (Theorem~\ref{thm:mvue} of Chapter~1).
\end{theorem}

\begin{example}
For a Bernoulli sample the sample proportion $\hat{P}$ has $\Var\hat{P}=p(1-p)/n=1/(n\Fish(p))$: it is efficient. For $\Normal(\mu,\sigma^2)$ the sample variance $S^2$ has $\Var S^2=2\sigma^4/(n-1)$ whereas the bound for unbiased estimators of $\sigma^2$ is $2\sigma^4/n$: $S^2$ is not efficient, but its efficiency $(n-1)/n\to1$.
\end{example}

\begin{pitfall}{when the bound does not apply}
The regularity conditions fail when the support of the distribution depends on the parameter. For the uniform distribution on $(0,\theta)$ the unbiased estimator $\frac{n+1}n\max X_i$ has variance $\theta^2/(n(n+2))$, of order $1/n^2$, \emph{smaller} than the order-$1/n$ value that the formal Cram\'er--Rao expression would give. There is no contradiction: the bound is not valid for this family.
\end{pitfall}

\section{Asymptotic properties of the maximum likelihood estimator}

\begin{theorem}[Asymptotic normality and efficiency of the MLE]\label{thm:mle-asym}
Under regularity, the MLE $\hat\Theta_n$ based on $n$ i.i.d.\ observations is consistent, and
\[ \sqrt n\big(\hat\Theta_n-\theta\big)\xrightarrow{d}\Normal\Big(0,\frac1{\Fish(\theta)}\Big). \]
So for large $n$, $\hat\Theta_n\approx\Normal\big(\theta,\ 1/(n\Fish(\theta))\big)$: the MLE attains the Cram\'er--Rao bound asymptotically.
\end{theorem}

\begin{proof}[Sketch]
Let $\ell(\theta)=\sum\ln f(X_i;\theta)$ and expand the likelihood equation $\ell'(\hat\theta)=0$ about the true $\theta$: $0=\ell'(\theta)+(\hat\theta-\theta)\ell''(\tilde\theta)$ for some $\tilde\theta$ between $\hat\theta$ and $\theta$. Hence
\[ \sqrt n(\hat\theta-\theta)=\frac{n^{-1/2}\ell'(\theta)}{-n^{-1}\ell''(\tilde\theta)}. \]
The numerator is $n^{-1/2}$ times a sum of i.i.d.\ scores with mean $0$ and variance $\Fish(\theta)$, so by the central limit theorem it tends to $\Normal(0,\Fish(\theta))$. The denominator tends in probability to $-\Exp[\partial^2\ln f]=\Fish(\theta)$ by the law of large numbers (consistency of $\hat\theta$ lets us replace $\tilde\theta$ by $\theta$). By Slutsky's theorem the ratio tends to $\Normal(0,\Fish(\theta)/\Fish(\theta)^2)=\Normal(0,1/\Fish(\theta))$.
\end{proof}

\begin{corollary}[Large-sample interval and Wald test]\label{cor:wald}
An approximate $100(1-\alpha)\%$ confidence interval for $\theta$ is $\hat\theta\pm z_{\alpha/2}/\sqrt{n\Fish(\hat\theta)}$. To test $H_0:\theta=\theta_0$ use $Z=(\hat\theta-\theta_0)\sqrt{n\Fish(\theta_0)}$ (or with $\hat\theta$ in place of $\theta_0$ in $\Fish$), approximately $\Normal(0,1)$ under $H_0$.
\end{corollary}

\begin{example}
For $n=200$ exponential observations with $\bar{x}=4$: $\hat\lambda=0.25$ and $\Fish(\lambda)=1/\lambda^2$, so $\se(\hat\lambda)=\hat\lambda/\sqrt n=0.25/14.14=0.0177$, with the $95\%$ interval $0.25\pm0.035$.
\end{example}

\begin{theorem}[Delta method]\label{thm:delta}
If $\sqrt n(\hat\theta-\theta)\to\Normal(0,v)$ and $g$ is differentiable with $g'(\theta)\ne0$, then $\sqrt n\big(g(\hat\theta)-g(\theta)\big)\to\Normal\big(0,\ g'(\theta)^2v\big)$.
\end{theorem}

\begin{proof}
$g(\hat\theta)-g(\theta)=g'(\tilde\theta)(\hat\theta-\theta)$ for some $\tilde\theta$ between them; as $\tilde\theta\to\theta$, $g'(\tilde\theta)\to g'(\theta)$; apply Slutsky.
\end{proof}

\begin{example}
The MLE of $\sigma$ in a normal sample is $\hat\sigma=\sqrt{\hat\sigma^2}$, where $\sqrt n(\hat\sigma^2-\sigma^2)\to\Normal(0,2\sigma^4)$. With $g(t)=\sqrt t$, $g'(\sigma^2)=1/(2\sigma)$, so $\sqrt n(\hat\sigma-\sigma)\to\Normal(0,\sigma^2/2)$ and $\se(\hat\sigma)\approx\sigma/\sqrt{2n}$.
\end{example}

\section{Sufficiency}

\begin{definition}[Sufficient statistic]
A statistic $T=T(X_1,\dots,X_n)$ is \emph{sufficient} for $\theta$ if the conditional distribution of the sample given $T$ does not depend on $\theta$: once $T$ is known, the rest of the data carry no further information about $\theta$.
\end{definition}

\begin{theorem}[Fisher--Neyman factorisation]\label{thm:factor}
$T$ is sufficient for $\theta$ if and only if the joint density factors as
\[ f(x_1,\dots,x_n;\theta)=g\big(T(\mathbf{x});\theta\big)\,h(\mathbf{x}), \]
where $h$ does not depend on $\theta$.
\end{theorem}

\begin{example}[Standard sufficient statistics]
\begin{itemize}
  \item Bernoulli$(p)$: $f=p^{\sum x_i}(1-p)^{n-\sum x_i}$, so $T=\sum X_i$ (the number of successes) is sufficient.
  \item Poisson$(\lambda)$: $f=e^{-n\lambda}\lambda^{\sum x_i}/\prod x_i!$, so $T=\sum X_i$.
  \item Exponential$(\lambda)$: $f=\lambda^ne^{-\lambda\sum x_i}$, so $T=\sum X_i$.
  \item $\Normal(\mu,\sigma^2)$: $\sum(x_i-\mu)^2=\sum(x_i-\bar{x})^2+n(\bar{x}-\mu)^2$ shows that $(\bar{X},S^2)$ is jointly sufficient for $(\mu,\sigma^2)$; for $\mu$ alone ($\sigma^2$ known) $\bar{X}$ is sufficient.
  \item Uniform$(0,\theta)$: $f=\theta^{-n}\mathbf{1}\{\max x_i\le\theta\}$, so $T=\max X_i$.
\end{itemize}
\end{example}

By the factorisation, the likelihood depends on the data only through $T$, so the MLE is a function of any sufficient statistic: in all the examples above the MLE is a function of $T$.

\begin{theorem}[Rao--Blackwell]\label{thm:rb}
Let $\hat\Theta$ be an unbiased estimator of $\theta$ and $T$ a sufficient statistic. Then $\tilde\Theta=\Exp[\hat\Theta\mid T]$ is an estimator (it does not depend on $\theta$, by sufficiency), is unbiased, and
\[ \Var(\tilde\Theta)\le\Var(\hat\Theta), \]
with equality only if $\hat\Theta$ is already a function of $T$.
\end{theorem}

\begin{proof}
Unbiasedness: $\Exp\tilde\Theta=\Exp\big[\Exp[\hat\Theta\mid T]\big]=\Exp\hat\Theta=\theta$. By the law of total variance, $\Var\hat\Theta=\Var(\Exp[\hat\Theta\mid T])+\Exp[\Var(\hat\Theta\mid T)]\ge\Var\tilde\Theta$.
\end{proof}

\begin{example}
For a Bernoulli sample, $\hat\Theta=X_1$ is an unbiased (and very poor) estimator of $p$. With $T=\sum X_i$: $\Exp[X_1\mid T]=T/n=\bar{X}$, the sample proportion, whose variance $p(1-p)/n$ is far smaller than $p(1-p)$. Conditioning on a sufficient statistic always improves an estimator.
\end{example}

\begin{definition}[Complete statistic]
A statistic $T$ is \emph{complete} if $\Exp_\theta[\phi(T)]=0$ for all $\theta$ implies $\phi(T)=0$ with probability one. For one-parameter exponential families (normal with known variance, Bernoulli, Poisson, exponential) the natural sufficient statistic is complete.
\end{definition}

\begin{theorem}[Lehmann--Scheff\'e]\label{thm:lehmann}
If $T$ is complete and sufficient and $\phi(T)$ is an unbiased estimator of $\theta$, then $\phi(T)$ is the unique MVUE of $\theta$.
\end{theorem}

\begin{proof}[Idea]
Take any unbiased estimator $\hat\Theta$; by Rao--Blackwell $\Exp[\hat\Theta\mid T]$ is a function of $T$ with no larger variance. Two unbiased functions of $T$ differ by a function with zero mean, which by completeness is zero. So the Rao--Blackwellised estimator is unique, and it has minimum variance.
\end{proof}

\begin{example}
For $X_i\sim\Normal(\mu,\sigma^2)$ the MVUE of $\mu$ is $\bar{X}$ and the MVUE of $\sigma^2$ is $S^2$ (both are unbiased functions of the complete sufficient statistic $(\bar{X},S^2)$), settling the questions of Chapter~1. For the uniform$(0,\theta)$ family, $\frac{n+1}n\max X_i$ is the MVUE of $\theta$.
\end{example}

\section{Bayes estimators and loss functions}

A \emph{loss function} $L(\hat\theta,\theta)$ measures the cost of estimating $\theta$ by $\hat\theta$; the standard choices are squared error $(\hat\theta-\theta)^2$ and absolute error $\abs{\hat\theta-\theta}$. The \emph{Bayes risk} of an estimator is the expected loss averaged over the prior and the data.

\begin{theorem}[Bayes estimators]\label{thm:bayes}
With a prior $\pi(\theta)$ and posterior $\pi(\theta\mid\mathbf{x})$, the estimator minimising the Bayes risk is, for squared-error loss, the \emph{posterior mean} $\Exp[\theta\mid\mathbf{x}]$; for absolute-error loss, the \emph{posterior median}; for $0$--$1$ loss, the \emph{posterior mode} (MAP estimate).
\end{theorem}

\begin{proof}[Squared error]
Minimise $\Exp[(\theta-a)^2\mid\mathbf{x}]$ over $a$; writing $(\theta-a)^2=(\theta-m)^2+2(\theta-m)(m-a)+(m-a)^2$ with $m=\Exp[\theta\mid\mathbf{x}]$, the cross term has zero posterior expectation, leaving $\Var(\theta\mid\mathbf{x})+(m-a)^2$, minimised at $a=m$.
\end{proof}

\begin{example}[Conjugacy and shrinkage]
For $X\sim\mathrm{Binomial}(n,p)$ and a $\mathrm{Beta}(a,b)$ prior the posterior is $\mathrm{Beta}(a+x,b+n-x)$, with mean $\dfrac{a+x}{a+b+n}$ (Example~\ref{ex:betabin}). It equals the weighted average $w\cdot\dfrac xn+(1-w)\cdot\dfrac a{a+b}$ with $w=\dfrac n{n+a+b}$. The Bayes estimator shrinks the MLE toward the prior mean; the prior acts like $a+b$ extra pseudo-observations. As $n\to\infty$, $w\to1$ and the estimator approaches the MLE, with variance approaching the Cram\'er--Rao bound; for small $n$ the shrinkage trades a little bias for a larger reduction in variance, which can lower the mean square error (compare Example~\ref{ex:mseexample}).
\end{example}

\begin{interview}{frequentist versus Bayesian in one paragraph}
A frequentist treats $\theta$ as a fixed unknown and judges an estimator by its sampling distribution (bias, variance, coverage of intervals). A Bayesian treats $\theta$ as random with a prior, and reports the posterior distribution; a $95\%$ \emph{credible interval} is an interval containing $\theta$ with posterior probability $0.95$, which is the statement people often mistakenly read into a confidence interval. For large samples with smooth priors the two approaches agree (the Bernstein--von Mises theorem: the posterior is approximately $\Normal(\hat\theta,1/(n\Fish(\hat\theta)))$), and for small samples the prior matters.
\end{interview}

\begin{summarybox}
\begin{itemize}
  \item Fisher information $\Fish(\theta)=\Var(\partial_\theta\ln f)=-\Exp[\partial_\theta^2\ln f]$ measures the curvature of the log-likelihood; it is additive: $n\Fish$ for a sample of size $n$.
  \item Cram\'er--Rao: every unbiased estimator has $\Var\ge1/(n\Fish(\theta))$; $\bar{X}$ (normal mean), $\hat{P}$ (proportion) and $\bar{X}$ (Poisson, exponential mean) attain it.
  \item The MLE is asymptotically $\Normal(\theta,1/(n\Fish))$: consistent, asymptotically unbiased and efficient; the delta method propagates this to functions of $\theta$.
  \item Sufficient statistics (factorisation theorem) retain all the information; Rao--Blackwell improves any unbiased estimator by conditioning on one; Lehmann--Scheff\'e gives the unique MVUE from a complete sufficient statistic.
  \item Bayes estimators are posterior summaries (mean for squared error); conjugate priors give closed forms and shrinkage toward the prior mean.
\end{itemize}
\end{summarybox}

\section{Exercises}

\begin{exercise}
Compute the Fisher information for (a) $\Normal(0,\sigma^2)$ with respect to $\sigma^2$, (b) the geometric distribution $f(x;p)=p(1-p)^{x-1}$, and give the Cram\'er--Rao bound for unbiased estimators of $p$ based on a sample of size $n$.
\end{exercise}

\begin{exercise}
Show that for a Poisson sample $\bar{X}$ attains the Cram\'er--Rao bound for estimating $\lambda$, and that $\bar{X}^2-\bar{X}/n$ is an unbiased estimator of $\lambda^2$. Is it efficient for $\lambda^2$?
\end{exercise}

\begin{exercise}
Prove the factorisation theorem in the discrete case (the easy direction: if the joint pmf factors as $g(T;\theta)h(\mathbf{x})$ then $T$ is sufficient).
\end{exercise}

\begin{exercise}
Show that $\sum X_i$ is sufficient for $\lambda$ in a Poisson sample, and use Rao--Blackwell to improve the estimator $\mathbf{1}\{X_1=0\}$ of $e^{-\lambda}$; find its improved form $\big(\frac{n-1}n\big)^{T}$.
\end{exercise}

\begin{exercise}
Using the delta method, find the approximate standard error of $\hat{p}/(1-\hat{p})$ (the odds) and of $\ln\big(\hat{p}/(1-\hat{p})\big)$ (the log-odds) for a binomial sample.
\end{exercise}

\begin{exercise}
In a Bayesian analysis of $n=50$ trials with $x=7$ successes and a $\mathrm{Beta}(1,1)$ (uniform) prior, find the posterior mean, the posterior mode, and compare with the MLE and with a Wald interval.
\end{exercise}

\begin{exercise}
For a $\Normal(\mu,\sigma^2)$ sample with known $\sigma^2$ and prior $\mu\sim\Normal(\mu_0,\tau^2)$, show that the Bayes estimator is $w\bar{x}+(1-w)\mu_0$ with $w=\dfrac{n/\sigma^2}{n/\sigma^2+1/\tau^2}$ and find its mean square error as a function of the true $\mu$. For which $\mu$ is it better than $\bar{X}$?
\end{exercise}

\begin{exercise}
Prove that the MLE of $\theta$ in the uniform$(0,\theta)$ family is biased and find the bias; show that the bias-corrected estimator has variance $\theta^2/(n(n+2))$.
\end{exercise}
