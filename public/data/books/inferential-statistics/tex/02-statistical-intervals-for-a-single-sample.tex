\chapter{Statistical Intervals for a Single Sample}

\begin{nuance}{how to use this chapter}
This chapter follows Chapter~8 of the course textbook section by section, with the same section titles and order. A point estimate gives one number and no sense of its reliability; an interval gives a range of plausible values together with a stated level of confidence. We develop confidence intervals for a mean (variance known and unknown), a variance, a proportion, and then prediction and tolerance intervals, which concern future observations rather than a parameter. Numbering follows the textbook with the chapter number lowered by $6$: Section $2.2$ of this book is Section $8$-$2$ of the textbook, and Section $2.2.1$ is its Section $8$-$2.1$.
\end{nuance}

\section{Introduction}

An estimate $\hat\theta$ will not equal $\theta$ exactly. A \emph{statistical interval} quantifies how far off it may be. Three kinds are used.

\begin{itemize}
  \item A \emph{confidence interval} is a range that contains a population parameter (such as $\mu$) with a stated probability, over repeated sampling.
  \item A \emph{prediction interval} is a range that contains a single future observation with a stated probability.
  \item A \emph{tolerance interval} is a range that contains at least a stated proportion of the population, with a stated confidence.
\end{itemize}

For the same data the three have different widths: the confidence interval is the narrowest (it concerns only the mean, whose uncertainty shrinks like $1/\sqrt n$), the prediction interval is wider (it must also include the variability of one new value), and the tolerance interval is the widest.

\section{Confidence Interval on the Mean of a Normal Distribution, Variance Known}

\subsection{Development of the Confidence Interval and Its Basic Properties}

Let $X_1,\dots,X_n$ be a random sample from $\Normal(\mu,\sigma^2)$ with $\sigma^2$ known and $\mu$ unknown. We know (Chapter~1) that
\[ Z=\frac{\bar{X}-\mu}{\sigma/\sqrt n}\sim\Normal(0,1). \]
Let $z_{\alpha/2}$ be the point that cuts off a right-tail area of $\alpha/2$ under the standard normal curve ($z_{0.025}=1.96$, $z_{0.05}=1.645$, $z_{0.005}=2.576$). Then $\Prob(-z_{\alpha/2}\le Z\le z_{\alpha/2})=1-\alpha$, and solving the inequalities for $\mu$:
\[ \Prob\Big(\bar{X}-z_{\alpha/2}\frac{\sigma}{\sqrt n}\ \le\ \mu\ \le\ \bar{X}+z_{\alpha/2}\frac\sigma{\sqrt n}\Big)=1-\alpha. \]

\begin{definition}[Confidence interval on the mean, $\sigma$ known]
If $\bar{x}$ is the mean of a random sample of size $n$ from a normal population with known variance $\sigma^2$, a $100(1-\alpha)\%$ confidence interval for $\mu$ is
\[ \bar{x}-z_{\alpha/2}\frac\sigma{\sqrt n}\ \le\ \mu\ \le\ \bar{x}+z_{\alpha/2}\frac\sigma{\sqrt n}. \]
The number $1-\alpha$ is the \emph{confidence level}; the two endpoints are the \emph{lower and upper confidence limits}.
\end{definition}

\begin{example}\label{ex:ci-known}
Ten measurements of the impact energy of a steel have $\bar{x}=64.46$ J, and $\sigma=1$ J is known. A $95\%$ confidence interval: $64.46\pm1.96\cdot1/\sqrt{10}=64.46\pm0.62$, that is $63.84\le\mu\le65.08$.
\end{example}

\begin{pitfall}{what ``95\% confidence'' means}
The $95\%$ refers to the \emph{procedure}, not to the particular interval. Before the data are collected, the random interval $\bar{X}\pm z_{\alpha/2}\sigma/\sqrt n$ will contain the fixed $\mu$ with probability $0.95$. After the data are in, the computed interval either contains $\mu$ or it does not; one cannot say ``the probability that $\mu$ lies in $(63.84,65.08)$ is $0.95$'' (in the frequentist framework $\mu$ is not random). The correct reading: if the procedure were repeated on many independent samples, about $95\%$ of the intervals would cover $\mu$.
\end{pitfall}

\subsection*{Properties}

\begin{itemize}
  \item \textbf{Width.} The length of the interval is $2z_{\alpha/2}\sigma/\sqrt n$, called the \emph{precision}. It shrinks with larger $n$ and larger $\alpha$ (lower confidence), and grows with larger $\sigma$ and higher confidence. The quantity $E=z_{\alpha/2}\sigma/\sqrt n$ is the \emph{error} (margin of error) in estimating $\mu$ by $\bar{x}$.
  \item \textbf{Confidence versus precision.} Greater confidence costs width: moving from $95\%$ to $99\%$ replaces $1.96$ by $2.576$, an increase of $31\%$ in the interval length.
\end{itemize}

\subsection{Choice of Sample Size}

With confidence $100(1-\alpha)\%$ we are sure that the error $\abs{\bar{x}-\mu}$ will not exceed $E=z_{\alpha/2}\sigma/\sqrt n$. Solving for $n$:

\begin{theorem}[Sample size for estimating a mean]\label{thm:samplesize-mean}
To be $100(1-\alpha)\%$ confident that the error in estimating $\mu$ by $\bar{x}$ does not exceed $E$, use
\[ n=\left(\frac{z_{\alpha/2}\,\sigma}{E}\right)^2,\qquad\text{rounded up to the next integer.} \]
\end{theorem}

\begin{example}
In Example~\ref{ex:ci-known}, how large must $n$ be for a $95\%$ error of at most $0.5$ J? $n=(1.96\cdot1/0.5)^2=15.37$, so $n=16$.
\end{example}

Because $n$ varies as $1/E^2$, halving the error multiplies the sample size by four.

\subsection{One-sided Confidence Bounds}

Sometimes only an upper or lower limit is of interest (a maximum emission level, a minimum strength).

\begin{definition}[One-sided confidence bounds]
A $100(1-\alpha)\%$ \emph{upper confidence bound} for $\mu$ is $\mu\le\bar{x}+z_\alpha\sigma/\sqrt n$, and a $100(1-\alpha)\%$ \emph{lower confidence bound} is $\bar{x}-z_\alpha\sigma/\sqrt n\le\mu$. Note $z_\alpha$, not $z_{\alpha/2}$: all of the error probability is placed in one tail.
\end{definition}

\begin{example}
Using the data above, a $95\%$ lower bound is $64.46-1.645/\sqrt{10}=63.94$, tighter than the two-sided lower limit $63.84$.
\end{example}

\subsection{General Method to Derive a Confidence Interval}

All the intervals in this chapter come from one idea.

\begin{definition}[Pivotal quantity]
A \emph{pivotal quantity} (or pivot) is a function $g(X_1,\dots,X_n;\theta)$ of the sample and the unknown parameter whose distribution does not depend on $\theta$ (or on any other unknown parameter).
\end{definition}

Recipe: (1) find a pivot $g$ for $\theta$ with a known distribution; (2) find constants $a,b$ with $\Prob(a\le g\le b)=1-\alpha$; (3) rearrange $a\le g(X;\theta)\le b$ into $L(X)\le\theta\le U(X)$.

\begin{example}
For $\Normal(\mu,\sigma^2)$ with $\sigma$ known, $g=(\bar{X}-\mu)/(\sigma/\sqrt n)$ is a pivot with a standard normal distribution; for $\mu$ with $\sigma$ unknown, $T=(\bar{X}-\mu)/(S/\sqrt n)\sim t_{n-1}$; for $\sigma^2$, $(n-1)S^2/\sigma^2\sim\chi^2_{n-1}$. The remaining sections apply the recipe to each.
\end{example}

\subsection{A Large-Sample Confidence Interval for $\mu$}

When the population is not normal or not assumed to be, but $n$ is large, the central limit theorem gives $\bar{X}\approx\Normal(\mu,\sigma^2/n)$. If $\sigma$ is unknown, replacing it by $S$ costs little because $S$ is close to $\sigma$ when $n$ is large.

\begin{theorem}[Large-sample interval]\label{thm:largesample}
If $n$ is large (say $n\ge40$), an approximate $100(1-\alpha)\%$ confidence interval for the mean $\mu$ of any population with finite variance is
\[ \bar{x}-z_{\alpha/2}\frac{s}{\sqrt n}\ \le\ \mu\ \le\ \bar{x}+z_{\alpha/2}\frac s{\sqrt n}. \]
\end{theorem}

\begin{example}
A sample of $n=50$ has $\bar{x}=23.4$ and $s=3.1$. A $90\%$ interval: $23.4\pm1.645\cdot3.1/\sqrt{50}=23.4\pm0.72$.
\end{example}

\subsection{Bootstrap Confidence Intervals (CD Only)}

For statistics such as the median or a correlation, no pivot with a known distribution is available. The bootstrap of Chapter~1 gives the \emph{percentile interval}.

\begin{definition}[Bootstrap percentile interval]
Generate $B$ bootstrap estimates $\hat\theta^*_1,\dots,\hat\theta^*_B$ by resampling the data with replacement and sort them. A $100(1-\alpha)\%$ percentile confidence interval is
\[ \big[\ \hat\theta^*_{(\alpha/2)},\ \hat\theta^*_{(1-\alpha/2)}\ \big], \]
the $\alpha/2$ and $1-\alpha/2$ sample quantiles of the bootstrap distribution. With $B=1000$ and $\alpha=0.05$ these are the $25$th and $975$th ordered values.
\end{definition}

A second version, the \emph{bootstrap standard-error interval}, is $\hat\theta\pm z_{\alpha/2}\se_{\mathrm{boot}}$ and is reasonable when the bootstrap distribution is roughly normal. The bootstrap requires no distributional assumption but needs a reasonably large and representative sample; see Chapter~13.

\section{Confidence Interval on the Mean of a Normal Distribution, Variance Unknown}

\subsection{The $t$ Distribution}

If $\sigma$ is unknown, substituting $S$ for $\sigma$ changes the sampling distribution of the pivot.

\begin{theorem}[The $t$ statistic]\label{thm:t-pivot}
If $X_1,\dots,X_n$ is a random sample from a \emph{normal} distribution with mean $\mu$, then
\[ T=\frac{\bar{X}-\mu}{S/\sqrt n} \]
has a $t$ distribution with $n-1$ degrees of freedom, $T\sim t_{n-1}$.
\end{theorem}

The $t$ density is bell-shaped and symmetric about $0$, like the standard normal, but with heavier tails; the shape depends on the degrees of freedom $k=n-1$. $\Var T=k/(k-2)$ for $k>2$, and as $k\to\infty$, $t_k\to\Normal(0,1)$. Let $t_{\alpha,k}$ denote the point with right-tail area $\alpha$ under $t_k$. For example $t_{0.025,9}=2.262$ and $t_{0.025,\infty}=1.960$.

\subsection{Development of the $t$ Distribution (CD Only)}

The statistic $T$ can be written
\[ T=\frac{(\bar{X}-\mu)/(\sigma/\sqrt n)}{\sqrt{S^2/\sigma^2}}=\frac{Z}{\sqrt{V/(n-1)}},\qquad Z\sim\Normal(0,1),\quad V=\frac{(n-1)S^2}{\sigma^2}\sim\chi^2_{n-1}, \]
with $Z$ and $V$ independent. This is the definition of the $t$ distribution with $n-1$ degrees of freedom. The independence of $\bar{X}$ and $S^2$ and the $\chi^2$ distribution of $V$ are proved in Chapter~11 (Theorem~\ref{thm:cochran}), where the density of $t_k$ is also written down.

\subsection{The $t$ Confidence Interval on $\mu$}

Using $\Prob(-t_{\alpha/2,n-1}\le T\le t_{\alpha/2,n-1})=1-\alpha$ and rearranging:

\begin{theorem}[$t$ interval]\label{thm:t-ci}
If $\bar{x}$ and $s$ are the mean and standard deviation of a random sample from a normal distribution with unknown variance $\sigma^2$, a $100(1-\alpha)\%$ confidence interval on $\mu$ is
\[ \bar{x}-t_{\alpha/2,n-1}\frac s{\sqrt n}\ \le\ \mu\ \le\ \bar{x}+t_{\alpha/2,n-1}\frac s{\sqrt n}. \]
One-sided bounds replace $t_{\alpha/2,n-1}$ by $t_{\alpha,n-1}$.
\end{theorem}

\begin{example}\label{ex:t-ci}
Ten measurements of the load at failure of a component have $\bar{x}=22.4$ kN and $s=3.0$ kN. A $95\%$ confidence interval: $t_{0.025,9}=2.262$, so $22.4\pm2.262\cdot3.0/\sqrt{10}=22.4\pm2.15$, that is $20.25\le\mu\le24.55$.
\end{example}

\begin{pitfall}{using $z$ with a small sample}
With $n=10$ the normal multiplier $1.96$ would give $22.4\pm1.86$, an interval that is too short and covers $\mu$ less than $95\%$ of the time. The $t$ multiplier corrects for the extra uncertainty from estimating $\sigma$. The correction is large for $n<30$ and negligible for $n>100$.
\end{pitfall}

The $t$ interval assumes a normal population; it is fairly robust to moderate non-normality for $n\ge15$ if the data are roughly symmetric, but is unreliable for small samples from skewed or heavy-tailed populations. Check with a normal probability plot.

\section{Confidence Interval on the Variance and Standard Deviation of a Normal Distribution}

Let $X_1,\dots,X_n$ be a random sample from $\Normal(\mu,\sigma^2)$ with sample variance $S^2$. The pivot is
\[ \chi^2=\frac{(n-1)S^2}{\sigma^2}\sim\chi^2_{n-1}. \]
Let $\chi^2_{\alpha,k}$ be the point with right-tail area $\alpha$. Then $\Prob\big(\chi^2_{1-\alpha/2,n-1}\le(n-1)S^2/\sigma^2\le\chi^2_{\alpha/2,n-1}\big)=1-\alpha$.

\begin{theorem}[Interval for $\sigma^2$ and $\sigma$]\label{thm:var-ci}
A $100(1-\alpha)\%$ confidence interval for $\sigma^2$ is
\[ \frac{(n-1)s^2}{\chi^2_{\alpha/2,n-1}}\ \le\ \sigma^2\ \le\ \frac{(n-1)s^2}{\chi^2_{1-\alpha/2,n-1}}, \]
and taking square roots gives an interval for $\sigma$. One-sided bounds: $\sigma^2\le(n-1)s^2/\chi^2_{1-\alpha,n-1}$ and $\sigma^2\ge(n-1)s^2/\chi^2_{\alpha,n-1}$.
\end{theorem}

\begin{example}
A sample of $n=20$ fill volumes has $s^2=0.0153$. For $95\%$: $\chi^2_{0.025,19}=32.85$ and $\chi^2_{0.975,19}=8.91$, so $19(0.0153)/32.85=0.00885\le\sigma^2\le19(0.0153)/8.91=0.0326$, and $0.094\le\sigma\le0.181$.
\end{example}

\begin{nuance}{asymmetry and non-normality}
Because the $\chi^2$ distribution is skewed, the interval is not symmetric about $s^2$ (the point estimate lies closer to the lower limit). And, unlike the $t$ interval for the mean, this interval is \emph{very sensitive} to non-normality, since the distribution of $S^2$ depends on the fourth moment of the population. It should be used only when normality is credible.
\end{nuance}

\section{A Large-Sample Confidence Interval for a Population Proportion}

Let $X$ be the number of successes in $n$ independent trials with success probability $p$; $\hat{P}=X/n$. For large $n$ the normal approximation to the binomial gives $\hat{P}\approx\Normal\big(p,\ p(1-p)/n\big)$, and replacing $p$ in the variance by $\hat{p}$:

\begin{theorem}[Interval for a proportion]\label{thm:prop-ci}
If $\hat{p}$ is the proportion of successes in a random sample of size $n$, an approximate $100(1-\alpha)\%$ confidence interval for $p$ is
\[ \hat{p}-z_{\alpha/2}\sqrt{\frac{\hat{p}(1-\hat{p})}n}\ \le\ p\ \le\ \hat{p}+z_{\alpha/2}\sqrt{\frac{\hat{p}(1-\hat{p})}n}. \]
The approximation is acceptable when $n\hat{p}$ and $n(1-\hat{p})$ are both at least about $5$ to $10$.
\end{theorem}

\begin{example}
In a sample of $n=85$ automobile engine crankshaft bearings, $10$ have rough surfaces. $\hat{p}=10/85=0.1176$, and a $95\%$ interval is $0.1176\pm1.96\sqrt{0.1176\cdot0.8824/85}=0.1176\pm0.0685$, that is $0.049\le p\le0.186$.
\end{example}

\begin{theorem}[Sample size for a proportion]\label{thm:samplesize-prop}
To be $100(1-\alpha)\%$ confident that the error $\abs{\hat{p}-p}$ is at most $E$, use
\[ n=\left(\frac{z_{\alpha/2}}E\right)^2p(1-p). \]
Since $p$ is unknown, use a preliminary estimate, or the worst case $p=0.5$ (which maximises $p(1-p)=0.25$), giving $n=0.25\,(z_{\alpha/2}/E)^2$.
\end{theorem}

\begin{example}
Poll sample size for a $\pm3\%$ margin at $95\%$ confidence with no prior information: $n=0.25(1.96/0.03)^2=1067.1$, so $n=1068$.
\end{example}

\begin{interview}{better intervals for proportions}
The interval of Theorem~\ref{thm:prop-ci} (the Wald interval) can have poor coverage when $p$ is near $0$ or $1$ or $n$ is small. The \emph{Wilson score interval}, obtained by solving $\abs{\hat{p}-p}\le z_{\alpha/2}\sqrt{p(1-p)/n}$ for $p$ exactly instead of plugging in $\hat{p}$, and the simple ``add two successes and two failures'' adjustment ($\tilde{p}=(x+2)/(n+4)$ in the formula above) both have much better coverage. For $x=0$ successes, the $95\%$ upper bound is approximately $3/n$ (the ``rule of three'').
\end{interview}

\section{A Prediction Interval for a Future Observation}

A confidence interval concerns a fixed parameter. Often we instead want a range for the next individual value, $X_{n+1}$, to be drawn from the same population.

\begin{theorem}[Prediction interval]\label{thm:pi}
Let $X_1,\dots,X_n$ be a random sample from $\Normal(\mu,\sigma^2)$ and $X_{n+1}$ an independent future observation. A $100(1-\alpha)\%$ prediction interval for $X_{n+1}$ is
\[ \bar{x}-t_{\alpha/2,n-1}\,s\sqrt{1+\frac1n}\ \le\ X_{n+1}\ \le\ \bar{x}+t_{\alpha/2,n-1}\,s\sqrt{1+\frac1n}. \]
\end{theorem}

\begin{proof}
$X_{n+1}-\bar{X}$ is normal with mean $0$ and variance $\sigma^2+\sigma^2/n=\sigma^2(1+1/n)$, because the new observation is independent of $\bar{X}$. Hence $\dfrac{X_{n+1}-\bar{X}}{S\sqrt{1+1/n}}=\dfrac{(X_{n+1}-\bar{X})/(\sigma\sqrt{1+1/n})}{\sqrt{S^2/\sigma^2}}\sim t_{n-1}$, and the interval follows as in Theorem~\ref{thm:t-ci}.
\end{proof}

\begin{example}
Continuing Example~\ref{ex:t-ci} ($n=10$, $\bar{x}=22.4$, $s=3.0$): the $95\%$ prediction interval for the load of the next component is $22.4\pm2.262\cdot3.0\sqrt{1.1}=22.4\pm7.12$, that is $15.3$ to $29.5$. The confidence interval for $\mu$, $20.25$ to $24.55$, is much narrower.
\end{example}

As $n\to\infty$ the prediction interval tends to $\mu\pm z_{\alpha/2}\sigma$, which does not shrink to zero: no amount of data removes the variability of a single future value. The confidence interval, by contrast, shrinks to a point.

\section{Tolerance Intervals for a Normal Distribution}

A \emph{tolerance interval} is meant to capture a stated proportion of the whole population, with a stated confidence that it really does.

\begin{definition}[Tolerance interval]
A \emph{tolerance interval for capturing at least $100\gamma\%$ of the values of a normal distribution with confidence level $100(1-\alpha)\%$} is
\[ \bar{x}-k\,s\ \le\ x\ \le\ \bar{x}+k\,s, \]
where $k=k(n,\gamma,\alpha)$ is a constant found from tables (or software) so that, over repeated samples, a proportion $1-\alpha$ of the resulting intervals contain at least a proportion $\gamma$ of the population.
\end{definition}

A good approximation for the two-sided factor (Howe's formula) is
\[ k\approx z_{(1+\gamma)/2}\sqrt{\frac{(n-1)\,(1+1/n)}{\chi^2_{1-\alpha,n-1}}}, \]
where $\chi^2_{1-\alpha,n-1}$ is the point with \emph{left} tail area $\alpha$ (right-tail area $1-\alpha$).

\begin{example}
For $n=10$, $\gamma=0.95$, $1-\alpha=0.95$: $z_{0.975}=1.96$, $\chi^2_{0.95,9}=3.325$, and $k\approx1.96\sqrt{9(1.1)/3.325}=1.96\sqrt{2.977}=3.38$; the exact tabled value is $3.38$. With $\bar{x}=22.4$, $s=3.0$ the tolerance interval is $22.4\pm3.38\cdot3.0$, i.e.\ $12.3$ to $32.5$, which is wider than the prediction interval $15.3$ to $29.5$.
\end{example}

\begin{interview}{three intervals, three questions}
\emph{Confidence interval:} where is the population mean? (narrowest, shrinks with $n$.) \emph{Prediction interval:} where will the next single observation fall? (limit $\mu\pm z\sigma$.) \emph{Tolerance interval:} what range covers $95\%$ of all units, with $95\%$ confidence that the claim is right? (widest; limit $\mu\pm z_{(1+\gamma)/2}\sigma$ from above.) Quality engineers use tolerance intervals for specification limits and confidence intervals for process averages.
\end{interview}

\begin{summarybox}
\begin{itemize}
  \item $\sigma$ known: $\bar{x}\pm z_{\alpha/2}\sigma/\sqrt n$; sample size $n=(z_{\alpha/2}\sigma/E)^2$; one-sided bounds use $z_\alpha$.
  \item $\sigma$ unknown (normal data): $\bar{x}\pm t_{\alpha/2,n-1}s/\sqrt n$; large samples (any population): $\bar{x}\pm z_{\alpha/2}s/\sqrt n$.
  \item Variance: $\dfrac{(n-1)s^2}{\chi^2_{\alpha/2,n-1}}\le\sigma^2\le\dfrac{(n-1)s^2}{\chi^2_{1-\alpha/2,n-1}}$ (sensitive to non-normality).
  \item Proportion: $\hat{p}\pm z_{\alpha/2}\sqrt{\hat{p}(1-\hat{p})/n}$; sample size $n=(z_{\alpha/2}/E)^2p(1-p)\le0.25(z_{\alpha/2}/E)^2$.
  \item Prediction interval: $\bar{x}\pm t_{\alpha/2,n-1}s\sqrt{1+1/n}$; tolerance interval: $\bar{x}\pm ks$.
  \item The confidence level describes the long-run coverage of the procedure, not the probability that one computed interval contains $\mu$.
\end{itemize}
\end{summarybox}

\section{Exercises}

\begin{exercise}
The tensile strength of a fiber is normal with $\sigma=2$ MPa. A sample of $9$ has mean $98.5$. Find $90\%$, $95\%$ and $99\%$ confidence intervals for $\mu$ and the sample size needed to estimate $\mu$ within $0.5$ MPa with $95\%$ confidence.
\end{exercise}

\begin{exercise}
A $95\%$ confidence interval for a mean based on $n=25$ and known $\sigma$ has length $4$. What would its length be for $n=100$? For $99\%$ confidence with $n=25$?
\end{exercise}

\begin{exercise}
Data on the time (minutes) for a chemical reaction: $\bar{x}=45.2$, $s=4.1$, $n=12$ from a normal population. Find the $95\%$ confidence interval for $\mu$, the $95\%$ upper confidence bound for $\mu$, and the $95\%$ confidence interval for $\sigma$.
\end{exercise}

\begin{exercise}
In a survey, $x=216$ of $n=400$ respondents prefer product A. Find the $95\%$ confidence interval for the proportion in the population. How large a sample would give a margin of error of $2\%$ with $99\%$ confidence?
\end{exercise}

\begin{exercise}
Explain, with a short numerical example, why a $95\%$ prediction interval is wider than a $95\%$ confidence interval from the same data, and compute both for $n=15$, $\bar{x}=50$, $s=6$.
\end{exercise}

\begin{exercise}
Use Howe's formula to compute the tolerance factor $k$ for $n=20$, $\gamma=0.99$, confidence $0.95$ (use $z_{0.995}=2.576$ and $\chi^2_{0.95,19}=10.12$).
\end{exercise}

\begin{exercise}
Show that the width of the large-sample confidence interval for a proportion is largest at $p=0.5$, and derive the sample size formula of Theorem~\ref{thm:samplesize-prop}.
\end{exercise}

\begin{exercise}
Draw $B=1000$ bootstrap samples from the data $4.1,\ 5.3,\ 3.9,\ 6.2,\ 4.8,\ 5.5,\ 4.4$ and form a bootstrap percentile $95\%$ interval for the median. Compare it with the $t$ interval for the mean.
\end{exercise}
